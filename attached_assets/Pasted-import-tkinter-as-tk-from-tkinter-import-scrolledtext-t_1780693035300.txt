import tkinter as tk
from tkinter import scrolledtext, ttk
import threading
import subprocess
import sys
import datetime

class SweptSentinelElite:
    def __init__(self, root):
        self.root = root
        self.root.title("S W E P T - S E N T I N E L | APEX OSINT UNIT")
        self.root.geometry("1300x900")
        self.root.configure(bg="#010810") # Professional Deep Blue-Black

        # --- HEADER (Clean Blue Style) ---
        self.header = tk.Frame(root, bg="#010810")
        self.header.pack(fill="x", pady=15)
        tk.Label(self.header, text="S W E P T - S E N T I N E L", font=("Consolas", 42, "bold"), fg="#00ccff", bg="#010810").pack()
        tk.Label(self.header, text="SYSTEM STATUS: ENCRYPTED NODE ACTIVE | 230 MODULES LOADED", font=("Courier", 10), fg="#004466", bg="#010810").pack()

        # --- TARGET INPUT SECTION ---
        self.input_frame = tk.Frame(root, bg="#0a1a2a", bd=1)
        self.input_frame.pack(pady=10, padx=100, fill="x")
        
        tk.Label(self.input_frame, text=" [TARGET_DATA] > ", font=("Consolas", 12), fg="#00ccff", bg="#0a1a2a").grid(row=0, column=0, padx=10, pady=15)
        self.target_entry = tk.Entry(self.input_frame, bg="black", fg="#00ffcc", insertbackground="cyan", width=65, font=("Consolas", 12), borderwidth=0)
        self.target_entry.grid(row=0, column=1, padx=5)

        # --- SCROLLABLE COMMAND GRID ---
        self.container = tk.Frame(root, bg="#010810")
        self.container.pack(fill="both", expand=True, padx=40)

        self.canvas = tk.Canvas(self.container, bg="#010810", highlightthickness=0)
        self.scrollbar = ttk.Scrollbar(self.container, orient="vertical", command=self.canvas.yview)
        self.scroll_frame = tk.Frame(self.canvas, bg="#010810")

        self.scroll_frame.bind("<Configure>", lambda e: self.canvas.configure(scrollregion=self.canvas.bbox("all")))
        self.canvas.create_window((0, 0), window=self.scroll_frame, anchor="nw")
        self.canvas.configure(yscrollcommand=self.scrollbar.set)
        self.canvas.pack(side="left", fill="both", expand=True)
        self.scrollbar.pack(side="right", fill="y")

        # --- FULL COMMAND DATASET (1-230) ---
        # Defining names for your specialized tools
        specialized = {
            "1": "IP TRACKER", "2": "DNS RESOLVE", "3": "PORT SCAN", "5": "WHOIS QUERY",
            "6": "PHONE OSINT", "7": "EMAIL REP", "8": "PROXY CHECK", "9": "DB SEARCH",
            "10": "GEOLOCATE", "45": "INSTAGRAM", "46": "TIKTOK", "49": "TELEGRAM ID",
            "53": "STEAM ID", "60": "TRUECALLER", "71": "VIN CHECK", "91": "SATELLITE",
            "131": "SQL MAP", "201": "BGP ROUTE", "207": "CDN ORIGIN", "230": "DMARC ANALYZE"
        }

        # Build full 1-230 list in order
        self.full_cmds = []
        for i in range(1, 231):
            num_str = str(i)
            name = specialized.get(num_str, f"MODULE_{i}")
            self.full_cmds.append((name, num_str))

        # Render 6-column Grid
        for i, (name, num) in enumerate(self.full_cmds):
            btn = tk.Button(
                self.scroll_frame, text=f"[{num}] {name}", width=21,
                bg="#0d1b2a", fg="#58a6ff", font=("Consolas", 8, "bold"),
                activebackground="#00ccff", activeforeground="black",
                relief="flat", pady=5, borderwidth=1,
                command=lambda n=num: self.execute(n)
            )
            btn.grid(row=i//6, column=i%6, padx=2, pady=2)

        # --- BOTTOM CONSOLE ---
        self.console = scrolledtext.ScrolledText(root, height=12, bg="black", fg="#00ff41", font=("Consolas", 10))
        self.console.pack(fill="x", padx=40, pady=20)
        self.log("SYSTEM READY. ENTER TARGET DATA AND SELECT A MODULE.")

    def log(self, text):
        now = datetime.datetime.now().strftime("%H:%M:%S")
        self.console.insert(tk.END, f"[{now}] {text}\n")
        self.console.see(tk.END)

    def execute(self, num):
        target = self.target_entry.get()
        if not target:
            self.log(f"ALERT: MODULE {num} REQUIRES TARGET DATA.")
            return
        
        self.log(f"EXECUTING MODULE {num} ON: {target}")
        threading.Thread(target=self.run_backend, args=(num, target), daemon=True).start()

    def run_backend(self, num, target):
        try:
            # Fixing the NameError by ensuring we call the actual python file
            process = subprocess.Popen(
                [sys.executable, 'swept_sentinel.py'], 
                stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, 
                text=True, bufsize=1
            )
            process.stdin.write(f"{num}\n{target}\n")
            process.stdin.flush()

            for line in iter(process.stdout.readline, ''):
                self.root.after(0, lambda l=line: self.console.insert(tk.END, l))
                self.root.after(0, lambda: self.console.see(tk.END))
            
            process.stdout.close()
            process.wait()
        except Exception as e:
            self.root.after(0, lambda: self.log(f"CRITICAL SYSTEM ERROR: {str(e)}"))

if __name__ == "__main__":
    root = tk.Tk()
    app = SweptSentinelElite(root)
    root.mainloop()
