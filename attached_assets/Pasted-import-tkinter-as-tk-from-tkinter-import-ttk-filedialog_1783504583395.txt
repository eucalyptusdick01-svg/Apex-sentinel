def generate_hash(data):
    hashed = hashlib.sha256(data.encode()).hexdigest()
    return str(int(hashed,16))[:9]

def generate_id(first,last,age,birth,state):

    code = state_codes.get(state,"00")

    while True:

        raw = f"{first}{last}{age}{birth}{state}{random.randint(1000,9999)}"
        core = generate_hash(raw)

        idnum = (code + core)[:9]

        if idnum not in id_set:
            id_set.add(idnum)
            return idnum

def add_record(record):

    generated_records.append(record)

    table.insert("", "end", values=(
        record["id"],
        record["first"],
        record["last"],
        record["state"]
    ))

    update_stats()

def update_stats():

    total = len(generated_records)

    states = len(set(r["state"] for r in generated_records))

    stats_label.config(
        text=f"Total IDs: {total} | States Used: {states}"
    )

def generate_single():

    first = first_name.get()
    last = last_name.get()
    age = age_entry.get()
    birth = birth_entry.get()
    state = state_box.get()

    if not first or not last or not state:
        messagebox.showerror("Error","Missing fields")
        return

    idnum = generate_id(first,last,age,birth,state)

    record = {
        "id":idnum,
        "first":first,
        "last":last,
        "state":state
    }

    add_record(record)

def load_csv():

    global data_rows

    path = filedialog.askopenfilename(filetypes=[("CSV","*.csv")])

    if not path:
        return

    with open(path) as f:
        reader = csv.DictReader(f)
        data_rows = list(reader)

    status_label.config(text=f"{len(data_rows)} records loaded")

def load_from_url():

    url = url_entry.get()

    if not url:
        return

    try:

        response = urllib.request.urlopen(url)
        lines = [l.decode("utf-8") for l in response.readlines()]
        reader = csv.DictReader(lines)

        global data_rows
        data_rows = list(reader)

        status_label.config(text=f"{len(data_rows)} records loaded from URL")

    except:
        messagebox.showerror("Error","Failed to load URL")

def generate_dataset():

    if not data_rows:
        messagebox.showerror("Error","No dataset loaded")
        return

    for row in data_rows:

        first = row.get("FirstName","")
        last = row.get("LastName","")
        age = row.get("Age","")
        birth = row.get("Birthdate","")
        state = row.get("State","")

        idnum = generate_id(first,last,age,birth,state)

        record = {
            "id":idnum,
            "first":first,
            "last":last,
            "state":state
        }

        add_record(record)

def bulk_generate():

    try:
        amount = int(bulk_entry.get())
    except:
        messagebox.showerror("Error","Invalid amount")
        return

    states = list(state_codes.keys())

    for i in range(amount):

        first = "User"+str(random.randint(100,999))
        last = "Test"+str(random.randint(100,999))
        age = random.randint(18,80)
        birth = random.randint(1950,2005)
        state = random.choice(states)

        idnum = generate_id(first,last,age,birth,state)

        record = {
            "id":idnum,
            "first":first,
            "last":last,
            "state":state
        }

        add_record(record)

def search_ids():

    query = search_entry.get()

    table.delete(*table.get_children())

    for r in generated_records:

        if query in r["id"]:
            table.insert("", "end", values=(r["id"],r["first"],r["last"],r["state"]))

def export_csv():

    path = filedialog.asksaveasfilename(defaultextension=".csv")

    if not path:
        return

    with open(path,"w",newline="") as f:

        writer = csv.writer(f)
        writer.writerow(["ID","FirstName","LastName","State"])

        for r in generated_records:
            writer.writerow([r["id"],r["first"],r["last"],r["state"]])

def export_json():

    path = filedialog.asksaveasfilename(defaultextension=".json")

    if not path:
        return

    with open(path,"w") as f:
        json.dump(generated_records,f,indent=4)

def export_db():

    path = filedialog.asksaveasfilename(defaultextension=".db")

    if not path:
        return

    conn = sqlite3.connect(path)
    c = conn.cursor()

    c.execute("""
    CREATE TABLE IF NOT EXISTS ids(
    id TEXT,
    firstname TEXT,
    lastname TEXT,
    state TEXT)
    """)

    for r in generated_records:

        c.execute(
        "INSERT INTO ids VALUES(?,?,?,?)",
        (r["id"],r["first"],r["last"],r["state"])
        )

    conn.commit()
    conn.close()
